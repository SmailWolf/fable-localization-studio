cmake_minimum_required(VERSION 3.20)
project(FableLocalizationStudio VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network Sql)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
add_executable(fable-localization-studio src/main.cpp src/babel.cpp src/fonts.cpp src/jobs.cpp src/tests.cpp)
target_link_libraries(fable-localization-studio PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Sql ZLIB::ZLIB Threads::Threads)
set_target_properties(fable-localization-studio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
