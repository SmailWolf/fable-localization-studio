cmake_minimum_required(VERSION 3.20)
project(FableLocalizationStudio VERSION 0.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network Sql)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
qt_add_executable(fable-localization-studio src/main.cpp src/babel.cpp src/fonts.cpp src/jobs.cpp src/tests.cpp)
target_link_libraries(fable-localization-studio PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network Qt6::Sql ZLIB::ZLIB Threads::Threads)
set_target_properties(fable-localization-studio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")

if(MSVC)
  target_compile_options(fable-localization-studio PRIVATE /utf-8)
  target_compile_definitions(fable-localization-studio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(WIN32)
  set_target_properties(fable-localization-studio PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
